# 실제 게임과 비교 장면이 같은 GPU 모듈을 연결합니다.
get_target_property(CASTLE_RAYLIB_SOURCE raylib SOURCE_DIR)
if(NOT EXISTS "${CASTLE_RAYLIB_SOURCE}/external/glad.h")
  message(FATAL_ERROR "고정한 raylib 소스의 OpenGL 로더를 찾지 못했습니다.")
endif()
add_library(castle_gpu STATIC gpu.c)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
foreach(stage common.glsl shadows.glsl terrain.glsl terrain.cs terrain.vs terrain.fs meadow.cs contacts.cs fields.cs geometry.vs geometry.fs grass.cs particles.cs smoke.vs smoke.fs smoke_composite.cs bins.cs light_bins.cs fog.cs cache.cs lighting.cs)
  set(path "${CMAKE_CURRENT_SOURCE_DIR}/${stage}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" shader)
  string(REPLACE "\\" "\\\\" shader "${shader}")
  string(REPLACE "\"" "\\\"" shader "${shader}")
  string(REPLACE "\n" "\\n\",\n\"" shader "${shader}")
  string(REPLACE "." "_" symbol "${stage}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/${symbol}.h"
    "static const char *const GPU_${symbol}[] = {\n\"${shader}\", 0};\n")
endforeach()
target_include_directories(castle_gpu PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(castle_gpu SYSTEM PRIVATE "${CASTLE_RAYLIB_SOURCE}")
target_link_libraries(castle_gpu PUBLIC raylib PRIVATE castle_frontend castle_warnings)
add_executable(castle_gpu_lab lab.c)
target_include_directories(castle_gpu_lab PRIVATE "${PROJECT_SOURCE_DIR}/core/src")
target_compile_definitions(castle_gpu_lab PRIVATE CASTLE_GPU_ROOT="${PROJECT_SOURCE_DIR}")
target_link_libraries(castle_gpu_lab PRIVATE castle_gpu castle_frontend castle_warnings)
if(MINGW)
  target_link_options(castle_gpu_lab PRIVATE -static)
endif()
install(TARGETS castle_gpu_lab RUNTIME DESTINATION . COMPONENT LightingDemo)
foreach(asset lighting_lab/pixel_albedo.png lighting_lab/pixel_normals.png lighting_lab/pixel_surface.png lighting_lab/scenery_masks.png lighting_lab/contact_policy.png terrain/biomes.png animals/walk.png roads/bridges.png fonts/Galmuri11.ttf)
  get_filename_component(parent "${asset}" DIRECTORY)
  install(FILES "${PROJECT_SOURCE_DIR}/game/assets/${asset}" DESTINATION "assets/${parent}" COMPONENT LightingDemo)
endforeach()
install(FILES "${PROJECT_SOURCE_DIR}/tools/lighting_lab/README_WINDOWS.md" DESTINATION . RENAME README.md COMPONENT LightingDemo)
install(FILES "${PROJECT_SOURCE_DIR}/CREDITS.md" DESTINATION . COMPONENT LightingDemo)
install(FILES "${PROJECT_SOURCE_DIR}/tools/lighting_lab/demo.cmd" "${PROJECT_SOURCE_DIR}/tools/lighting_lab/shadow-demo.cmd" "${PROJECT_SOURCE_DIR}/tools/lighting_lab/check.cmd" "${PROJECT_SOURCE_DIR}/tools/lighting_lab/benchmark.cmd" "${PROJECT_SOURCE_DIR}/tools/lighting_lab/benchmark_gpu.ps1" DESTINATION . COMPONENT LightingDemo)
install(FILES "${PROJECT_SOURCE_DIR}/game/assets/licenses/Galmuri-OFL.txt" "${PROJECT_SOURCE_DIR}/game/assets/licenses/generated-art.txt" DESTINATION assets/licenses COMPONENT LightingDemo)
install(FILES "${CASTLE_RAYLIB_SOURCE}/../LICENSE" DESTINATION licenses RENAME raylib.txt COMPONENT LightingDemo)
install(FILES "${CASTLE_RAYLIB_SOURCE}/external/glfw/LICENSE.md" DESTINATION licenses RENAME GLFW.txt COMPONENT LightingDemo)
