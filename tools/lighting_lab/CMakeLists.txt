# 별도 실행 파일이므로 실제 게임의 렌더링 경로를 바꾸지 않습니다.
add_executable(castle_lighting_lab main.c ../art/lighting_lab.c ../art/trace.c)
add_executable(castle_pixel_lighting pixel_main.c ../art/lighting_pixels.c ../art/trace.c)
target_include_directories(castle_pixel_lighting PRIVATE ../art)
target_link_libraries(castle_pixel_lighting PRIVATE castle_frontend raylib castle_warnings)
target_include_directories(castle_lighting_lab PRIVATE ../art "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(castle_lighting_lab PRIVATE CASTLE_LIGHTING_ROOT="${PROJECT_SOURCE_DIR}")
target_link_libraries(castle_lighting_lab PRIVATE castle_frontend raylib glfw castle_warnings)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
foreach(shader gbuffer lighting)
  set(path "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.fs")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" source)
  string(REPLACE "\\" "\\\\" source "${source}")
  string(REPLACE "\"" "\\\"" source "${source}")
  string(REPLACE "\n" "\\n\",\n\"" source "${source}")
  string(TOUPPER "${shader}" upper)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/${shader}.h"
    "static const char *const LAB_${upper}_LINES[] = {\n\"${source}\", 0};\n")
endforeach()

  find_program(CASTLE_NODE_EXECUTABLE NAMES node nodejs)
  if(CASTLE_NODE_EXECUTABLE)
    add_test(NAME pixel_lighting_shadows COMMAND "${CASTLE_NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/check_pixel_shadows.mjs")
  endif()
endif()
