# 화면과 분리된 C 실행 계층은 그래픽 라이브러리 없이도 테스트합니다.
add_library(castle_frontend STATIC src/runtime/path.c src/runtime/session.c src/runtime/combat.c src/runtime/storage.c src/render/sprite_layout.c src/render/effects.c src/runtime/loop.c)
target_include_directories(castle_frontend PUBLIC src PRIVATE ${PROJECT_SOURCE_DIR}/core/src)
target_link_libraries(castle_frontend PUBLIC castle_core_static PRIVATE castle_warnings)
if(NOT WIN32)
  target_link_libraries(castle_frontend PUBLIC m)
else()
  target_link_libraries(castle_frontend PUBLIC shell32)
endif()

if(CASTLE_BUILD_GAME)
  include(${PROJECT_SOURCE_DIR}/cmake/Data.cmake)
  # 컴퓨트 지원 여부가 모호한 설치본 대신 고정한 소스를 같은 설정으로 빌드합니다.
  if(NOT TARGET raylib)
    include(FetchContent)
    if(POLICY CMP0135)
      cmake_policy(SET CMP0135 NEW)
    endif()
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
    set(SUPPORT_FILEFORMAT_MOD OFF CACHE BOOL "" FORCE)
    set(SUPPORT_FILEFORMAT_XM OFF CACHE BOOL "" FORCE)
    set(SUPPORT_MODULE_RMODELS OFF CACHE BOOL "" FORCE)
    set(SUPPORT_CLIPBOARD_IMAGE OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
    set(OPENGL_VERSION "4.3" CACHE STRING "" FORCE)
    set(CASTLE_RAYLIB_URL "https://codeload.github.com/raysan5/raylib/tar.gz/refs/tags/5.5")
    if(EXISTS "${PROJECT_SOURCE_DIR}/third_party/raylib-5.5.tar.gz")
      set(CASTLE_RAYLIB_URL "${PROJECT_SOURCE_DIR}/third_party/raylib-5.5.tar.gz")
    endif()
    FetchContent_Declare(raylib URL "${CASTLE_RAYLIB_URL}"
      URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e)
    FetchContent_MakeAvailable(raylib)
    # 5.5의 설정 파서가 조건부 매크로 양쪽(7/9)을 모두 등록합니다.
    # 켜진 GPU 스키닝 분기의 9를 남겨 외부 코드의 중복 정의를 피합니다.
    get_target_property(CASTLE_RAYLIB_DEFS raylib COMPILE_DEFINITIONS)
    if("MAX_MESH_VERTEX_BUFFERS=9" IN_LIST CASTLE_RAYLIB_DEFS)
      list(FILTER CASTLE_RAYLIB_DEFS EXCLUDE REGEX "^MAX_MESH_VERTEX_BUFFERS=7$")
      set_target_properties(raylib PROPERTIES COMPILE_DEFINITIONS "${CASTLE_RAYLIB_DEFS}")
    endif()
    if(MSVC)
      target_compile_options(raylib PRIVATE /utf-8)
      if(TARGET glfw)
        target_compile_options(glfw PRIVATE /utf-8)
      endif()
    endif()
  endif()
  add_subdirectory("${PROJECT_SOURCE_DIR}/tools/lighting_lab/gpu" "${PROJECT_BINARY_DIR}/graphics_gpu")
  add_executable(castle_game src/main.c src/ui/widgets.c src/ui/panels.c src/render/world.c src/assets/loader.c src/ui/residents.c src/ui/services.c src/ui/facilities.c
    src/ui/main_menu.c src/ui/tutorial.c)
  target_sources(castle_game PRIVATE src/render/gpu_bridge.c src/render/carried_lights.c src/render/scene_profiles.c
                                     src/review.c)
  set(CASTLE_GAME_OUTPUT_DIR "" CACHE PATH "실행 중인 게임과 분리할 선택적 검증 출력 폴더")
  if(CASTLE_GAME_OUTPUT_DIR)
    set_target_properties(castle_game PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CASTLE_GAME_OUTPUT_DIR}")
  endif()
  target_include_directories(castle_game PRIVATE ${PROJECT_SOURCE_DIR}/core/src ${PROJECT_SOURCE_DIR}/cli)
  target_link_libraries(castle_game PRIVATE castle_frontend castle_cli_support castle_gpu raylib castle_warnings)
  if(MINGW)
    target_link_options(castle_game PRIVATE -static)
  endif()
  target_link_libraries(castle_game PRIVATE castle_asset_index)
  add_custom_target(castle_game_assets ALL
    COMMAND ${CMAKE_COMMAND} -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/assets -DDESTINATION=$<TARGET_FILE_DIR:castle_game>/assets -DMANIFEST=${CASTLE_ASSET_MANIFEST} -P ${PROJECT_SOURCE_DIR}/cmake/CopyAssets.cmake)
  add_dependencies(castle_game castle_game_assets)
  install(TARGETS castle_game RUNTIME DESTINATION . COMPONENT Runtime)
  foreach(asset IN LISTS CASTLE_RUNTIME_ASSET_PATHS)
    get_filename_component(parent "${asset}" DIRECTORY)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/assets/${asset}" DESTINATION "assets/${parent}" COMPONENT Runtime)
  endforeach()
  install(FILES "${CASTLE_ASSET_MANIFEST}" DESTINATION assets COMPONENT Runtime)
  foreach(data_file IN LISTS CASTLE_DATA_FILES)
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/data/${data_file}" DESTINATION data COMPONENT Runtime)
  endforeach()
  install(FILES ${PROJECT_SOURCE_DIR}/CREDITS.md DESTINATION . COMPONENT Runtime)
  install(FILES ${PROJECT_SOURCE_DIR}/docs/PLAYER_GUIDE.md DESTINATION . RENAME README.md COMPONENT Runtime)
  install(FILES ${PROJECT_SOURCE_DIR}/docs/GRAPHICS_INTEGRATION.md DESTINATION . RENAME GRAPHICS.md COMPONENT Runtime)
  install(FILES ${PROJECT_SOURCE_DIR}/docs/art/game_graphics.png ${PROJECT_SOURCE_DIR}/docs/art/game_graphics_tutorial.png DESTINATION art COMPONENT Runtime)
  if(WIN32)
    install(FILES ${PROJECT_SOURCE_DIR}/tools/graphics/start-game.cmd ${PROJECT_SOURCE_DIR}/tools/graphics/check-graphics.cmd DESTINATION . COMPONENT Runtime)
  endif()
  if(DEFINED raylib_SOURCE_DIR)
    install(FILES "${raylib_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME raylib.txt COMPONENT Runtime)
    install(FILES "${raylib_SOURCE_DIR}/src/external/glfw/LICENSE.md" DESTINATION licenses RENAME GLFW.txt COMPONENT Runtime)
  endif()
  if(CASTLE_BUILD_TESTS)
    add_test(NAME native_smoke_20 COMMAND castle_game --smoke-test --days 20 --seed 42 --save-dir "${CMAKE_CURRENT_BINARY_DIR}/smoke_saves")
    set_tests_properties(native_smoke_20 PROPERTIES TIMEOUT 600)
    add_test(NAME native_bad_arguments COMMAND castle_game --days abc)
    if(CASTLE_GPU_TESTS)
      add_test(NAME gpu_menu_transitions COMMAND castle_game --gpu-transitions 3 --no-audio --width 800 --height 600)
      set_tests_properties(gpu_menu_transitions PROPERTIES TIMEOUT 600 RUN_SERIAL TRUE LABELS gpu)
    endif()
    set_tests_properties(native_bad_arguments PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
